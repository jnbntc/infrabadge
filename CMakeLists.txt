cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3 CACHE STRING "InfraBadge V2 target")
option(IB_BUILD_M2 "Opt in to network dependency closure for the separate M2 build" OFF)
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(IB_BUILD_M2 "${IB_BUILD_M2}")
project(infrabadge VERSION 0.1.0)
if(IB_BUILD_M2 AND NOT CONFIG_IB_M2_CONNECTIVITY)
    message(FATAL_ERROR "IB_BUILD_M2 requires the M2 profile")
endif()
if(CONFIG_IB_M2_CONNECTIVITY AND NOT IB_BUILD_M2)
    message(FATAL_ERROR "M2 requires -DIB_BUILD_M2=ON in its own build directory")
endif()
idf_build_get_property(infrabadge_idf_version IDF_VER)
if(NOT infrabadge_idf_version STREQUAL "v5.5.5")
    message(FATAL_ERROR "InfraBadge M0 requires ESP-IDF v5.5.5; found ${infrabadge_idf_version}")
endif()
