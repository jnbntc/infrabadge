cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3 CACHE STRING "InfraBadge V2 target")
option(IB_BUILD_M2 "Opt in to network dependency closure for the separate M2 build" OFF)
option(IB_BUILD_M2_AUTO "Include the separate autonomous validation application" OFF)
option(IB_BUILD_M35 "Include M3.5 multisite UI" OFF)
option(IB_BUILD_M36 "Include roaming and night-mode application" OFF)
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(IB_BUILD_M2 "${IB_BUILD_M2}")
idf_build_set_property(IB_BUILD_M2_AUTO "${IB_BUILD_M2_AUTO}")
idf_build_set_property(IB_BUILD_M35 "${IB_BUILD_M35}")
idf_build_set_property(IB_BUILD_M36 "${IB_BUILD_M36}")
project(infrabadge VERSION 0.1.0)
if(IB_BUILD_M35 AND (NOT IB_BUILD_M2 OR NOT CONFIG_IB_M35_UI OR IB_BUILD_M2_AUTO))
    message(FATAL_ERROR "M3.5 requires its separate network/UI profile")
endif()
if(CONFIG_IB_M35_UI AND NOT IB_BUILD_M35)
    message(FATAL_ERROR "Use -DIB_BUILD_M35=ON for M3.5")
endif()
if(IB_BUILD_M2_AUTO AND (NOT IB_BUILD_M2 OR NOT CONFIG_IB_M2_AUTO_DIAG))
    message(FATAL_ERROR "Autonomous diagnostics require their separate M2 build/defaults")
endif()
if(CONFIG_IB_M2_AUTO_DIAG AND NOT IB_BUILD_M2_AUTO)
    message(FATAL_ERROR "Use -DIB_BUILD_M2_AUTO=ON for the diagnostic profile")
endif()
if(IB_BUILD_M2 AND NOT CONFIG_IB_M2_CONNECTIVITY)
    message(FATAL_ERROR "IB_BUILD_M2 requires the M2 profile")
endif()
if(CONFIG_IB_M2_CONNECTIVITY AND NOT IB_BUILD_M2)
    message(FATAL_ERROR "M2 requires -DIB_BUILD_M2=ON in its own build directory")
endif()
idf_build_get_property(infrabadge_idf_version IDF_VER)
if(NOT infrabadge_idf_version STREQUAL "v5.5.5")
    message(FATAL_ERROR "InfraBadge M0 requires ESP-IDF v5.5.5; found ${infrabadge_idf_version}")
endif()

if(IB_BUILD_M36 AND (NOT IB_BUILD_M35 OR NOT CONFIG_IB_M36_ROAMING))
    message(FATAL_ERROR "M3.6 requires its separate profile and M3.5 UI")
endif()
if(CONFIG_IB_M36_ROAMING AND NOT IB_BUILD_M36)
    message(FATAL_ERROR "Use -DIB_BUILD_M36=ON for roaming")
endif()
